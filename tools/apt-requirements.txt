# tools/apt-requirements.txt -- Debian/Ubuntu packages needed to build and
# run the project tools. One package per line. A line "# [tool]" starts
# the group for tools/check_<tool>.sh; a package may appear in several
# groups. Everything after # is a comment.
#
# The scripts never install packages. Install all of them with:
#   sudo apt install $(sed 's/#.*//' tools/apt-requirements.txt)
